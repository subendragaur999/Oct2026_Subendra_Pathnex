* Ansible -Install  & Enable  Nginx 

  - name: install and start nginx on pathnex

  host: all
  become: yes 

  task:
   - name: install nginx
   yum:
   name: nginx
  state: present	

   - name: enable nginx
  service: 
     name: nginx
    state: started
   enable: yes

  name: pathnex-deployment

  replicas: 2
  selector:
    matchLabels:
      app: pathnex-app
  template:
    metadata:
      labels:
        app: pathnex-app
    spec:
      containers:
        - name: app
